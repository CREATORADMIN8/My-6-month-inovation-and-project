async def speak(text):
    speaker = win.Dispatch("SAPI.SpVoice")
    speaker.Rate = -2
    speaker.Speak(text)

async def listen():
    r= SR.Recognizer()
    r.energy_threshold = 200
    r.dynamic_energy_threshold = True
    mic = SR.Microphone()
    with mic as source:
      r.adjust_for_ambient_noise(source, duration=0.5)
      try:
         audio= r.listen(source, timeout=60, phrase_time_limit=4)
         text= r.recognize_google(audio).lower()
         return text
      except SR.WaitTimeoutError:
         return "sleep"
      except SR.UnknownValueError:
          return ""
      except Exception:
          return ""

async def load_memory():
   if os.path.exists(MEMORY):
      try:
         with open(MEMORY, "r", encoding="utf-8") as f:
            return json.load(f)
      except Exception:
         return[]
   return []

async def save_history(topic, user_text, ai_reply):
   memory= await load_memory()
   memory.append({
      "topic": topic,
      "text": user_text,
      "data": ai_reply
   })
   with open(MEMORY, "w", encoding="utf-8") as f:
      json.dump(memory, f, indent=4)

async def aalu(websocket):
     await speak("orvok system activated")
     while True: 
         text = await listen()
         if "orvok" in text:
            await speak("hey whatsapp bro what help you want")
            text = await listen()
         elif text == "":
            await speak("I cant't hear you please speak clearly or check your connection")
            continue
         elif text == "sleep":
            await speak("orvok is now in sleep mode")
            continue
         else:
            await speak("please call me with my name orvok")
            continue
         check_data= await check(text)
         if check_data.lower() == "true":
            model = "gemini-2.5-flash"
            config = types.LiveConnectConfig(response_modalities=["TEXT"])
            async with api.aio.live.connect(model=model, config=config) as session:
              await speak("gemini live session active")
              video_task = asyncio.create_task(stream_video_to_gemini(session, text, websocket))
              try:
                  async for response in session.receive():
                    server_content = response.server_content
                    if server_content and server_content.model_turn:
                      for part in server_content.model_turn.parts:
                         if part.text:
                           await speak(part.text)
              finally:
                  video_task.cancel()
                  await asyncio.gather(video_task, return_exceptions=True)
         else:
            data= api.models._generate_content(
               model="gemini-3.5-flash-lite",
               contents= [
                  (
                    f"the user talking about which topic in ({text}), like ( normal, innovator, builder, maker, health care, gym trainer/ workout guide, coder, planner, time manager, care taker, artist, acedemic, psycology analysist, cooking , advisor) or none of these"
                  )
               ],
            )
            mode= data.text.strip()
            ai_reply= await get(mode, text, safety_settings)
            await save_history(mode, text, ai_reply)
            await speak(ai_reply)

async def orvok(websocket):
   CONECTED_CLIENTS.add(websocket)
   try:
      async for message in websocket:
        data= json.loads(message)
        msg= data.get("type")

        if msg in ["chatq", "geminq"]:
           mode = data.get("mode", "normal")
           query = data.get("query", "")

           if "projection" in query.lower() or "3d" in query.lower():
               html_path= os.path.abspath("sketchfab_viewer.html")
               if os.path.exists(html_path):
                  webbrowser.open(f"file://{html_path}")

           ai_reply= await get(mode, query, safety_settings)
           await websocket.send(json.dumps({
              "type": "chat_response",
              "text": ai_reply
           }))
   except websockets.exceptions.ConnectionClosed:
      pass
   finally:
      CONECTED_CLIENTS.remove(websocket)

   await speak("server started on ws://localhost:8765")
   await asyncio.gather(aalu(), server.wait_closed())
